import os import requests import json import speech_recognition as sr import google.generativeai as genai from googlesearch import search from github import Github # अपनी असली API Keys यहाँ डालें GEMINI_API_KEY = "YOUR_FREE_GEMINI_API_KEY" GITHUB_TOKEN = "YOUR_GITHUB_PERSONAL_ACCESS_TOKEN" GITHUB_REPO_NAME = "YOUR_GITHUB_USERNAME/YOUR_REPO_NAME" genai.configure(api_key=GEMINI_API_KEY) model = genai.GenerativeModel('gemini-1.5-flash') TASKER_URL = "http://localhost:1821/JarvisAction" def speak(text): print(f"Jarvis: {text}") try: requests.post(TASKER_URL, data={'action': 'speak', 'text': text}) except: pass def listen_command(): r = sr.Recognizer() with sr.Microphone() as source: print("सुन रहा हूँ सर...") r.pause_threshold = 1 audio = r.listen(source) try: query = r.recognize_google(audio, language='hi-IN') print(f"Sir: {query}\n") return query.lower() except: return "none" def send_to_phone(action_type, target): try: requests.post(TASKER_URL, json={'action': action_type, 'target': target}) except: print("टूल अलर्ट: टास्कर कनेक्ट नहीं है।") def google_search_tool(query): speak(f"गूगल पर '{query}' सर्च कर रहा हूँ सर।") search_results = [] try: for url in search(query, num_results=3): search_results.append(url) return f"गूगल पर ये लिंक्स मिले हैं: {', '.join(search_results)}" except Exception as e: return "गूगल सर्च में कुछ दिक्कत आई।" def deploy_to_github_tool(filename, file_content): speak(f"सर, मैं {filename} कोड को बनाकर आपके गिटहब पर लाइव अपलोड कर रहा हूँ।") try: g = Github(GITHUB_TOKEN) repo = g.get_user().get_repo(GITHUB_REPO_NAME.split('/')[-1]) try: contents = repo.get_contents(filename) repo.update_file(contents.path, "Jarvis Auto Update", file_content, contents.sha) status = f"सर, मैंने गिटहब पर {filename} को अपडेट कर दिया है।" except: repo.create_file(filename, "Jarvis Auto Deploy", file_content) status = f"सर, मैंने गिटहब पर नई फाइल {filename} बनाकर अपलोड कर दी है।" return status except Exception as e: return f"माफ़ कीजिएगा सर, गिटहब पर अपलोड करने में त्रुटि आई: {str(e)}" def process_jarvis_logic(user_command): system_prompt = ( f"You are Jarvis, an advanced AI supervisor. Analyze the user command: '{user_command}'. " "Decide which tool to use. Respond ONLY in a JSON format. " "Options:\n" "1. For google search: {{'tool': 'google', 'query': 'search terms'}}\n" "2. For making and deploying code to web/github: {{'tool': 'deploy_code', 'filename': 'index.html', 'content': 'complete code structure here'}}\n" "3. For phone actions (call/message/app): {{'tool': 'phone', 'action': 'open_app/make_call', 'target': 'name'}}\n" "4. For general talk: {{'tool': 'chat', 'reply': 'your response'}}" ) try: ai_response = model.generate_content(system_prompt) clean_text = ai_response.text.strip().replace("```json", "").replace("```", "") decision = json.loads(clean_text) tool = decision.get('tool') if tool == "google": result = google_search_tool(decision.get('query')) speak(result) elif tool == "deploy_code": result = deploy_to_github_tool(decision.get('filename'), decision.get('content')) speak(result) elif tool == "phone": action = decision.get('action') target = decision.get('target') speak(f"फोन टास्क प्रोसेस कर रहा हूँ सर।") send_to_phone(action, target) else: normal_reply = model.generate_content(user_command) speak(normal_reply.text) except Exception as e: try: normal_reply = model.generate_content(user_command) speak(normal_reply.text) except: speak("सर्वर रिस्पॉन्स में थोड़ी दिक्कत है सर।") if __name__ == "__main__": speak("सिस्टम पूरी तरह अपग्रेड हो चुका है सर। लाइव डिप्लॉयमेंट टूल्स तैयार हैं।") while True: command = listen_command() if command == "none": continue if "offline" in command or "बंद हो जाओ" in command: speak("ऑफलाइन हो रहा हूँ सर।") break process_jarvis_logic(command)
