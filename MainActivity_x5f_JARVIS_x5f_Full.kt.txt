@Entity(tableName = "shers")
data class SherEntity(@PrimaryKey(autoGenerate = true) val id: Long = 0, val poet: String, val textUrduHindi: String, val theme: String = "khet", var likeCount: Int = 0, var isLiked: Boolean = false)
@Entity(tableName = "products")
data class ProductEntity(@PrimaryKey val id: String, val name: String, val rate: Double, val photoUri: String, val timestamp: Long = System.currentTimeMillis())
@Dao
interface SherDao {
    @Query("SELECT * FROM shers ORDER BY id ASC") fun getAllShers(): Flow<List<SherEntity>>
    @Insert(onConflict = OnConflictStrategy.REPLACE) suspend fun insertAll(shers: List<SherEntity>)
    @Insert(onConflict = OnConflictStrategy.REPLACE) suspend fun insert(sher: SherEntity)
    @Update suspend fun update(sher: SherEntity)
    @Query("SELECT COUNT(*) FROM shers") suspend fun getCount(): Int
}
@Dao
interface ProductDao {
    @Query("SELECT * FROM products ORDER BY timestamp DESC") fun getAllProducts(): Flow<List<ProductEntity>>
    @Insert(onConflict = OnConflictStrategy.REPLACE) suspend fun insert(product: ProductEntity)
}
@Database(entities = [SherEntity::class, ProductEntity::class], version = 2, exportSchema = false)
abstract class AzadKahaniDatabase : RoomDatabase() {
    abstract fun sherDao(): SherDao
    abstract fun productDao(): ProductDao
    companion object {
        @Volatile private var INSTANCE: AzadKahaniDatabase? = null
        fun getInstance(context: Context): AzadKahaniDatabase =
            INSTANCE ?: synchronized(this) {
                INSTANCE ?: Room.databaseBuilder(context.applicationContext, AzadKahaniDatabase::class.java, "azad_kahani_jarvis.db").fallbackToDestructiveMigration().build().also { INSTANCE = it }
            }
    }
}
data class ProphetModel(val id: Int, val name: String, val arabicName: String, val pages: List<String>)
data class QuranVerseOrPage(val pageNumber: Int, val surahName: String, val arabic: String, val urdu: String, val hindi: String, val english: String, val marathi: String, val punjabi: String)
data class ScientistModel(val id: Int, val name: String, val era: String, val field: String, val contribution: String)

object AssetDataRepository {
    private fun readAsset(context: Context, filename: String): String? = try { context.assets.open(filename).bufferedReader().use { it.readText() } } catch (_: Exception) { null }
    fun loadShers(context: Context): List<SherEntity> {
        val json = readAsset(context, "shers_500.json")
        if (json != null) { try { val type = object : TypeToken<List<SherEntity>>() {}.type; return Gson().fromJson(json, type) } catch (_: Exception) {} }
        return (1..500).map { i -> SherEntity(id = i.toLong(), poet = if (i%2==0) "Allama Iqbal" else "Mirza Ghalib", textUrduHindi = if (i%2==0) "सितारों से आगे जहाँ और भी हैं\nअभी इश्क़ के इम्तिहाँ और भी हैं #"+i else "हज़ारों ख़्वाहिशें ऐसी कि हर ख़्वाहिश पे दम निकले\nबहुत निकले मिरे अरमान लेकिन फिर भी कम निकले #"+i, theme = if (i%2==0) "khet" else "paani", likeCount = i*3) }
    }
    fun loadProphets(context: Context): List<ProphetModel> {
        val json = readAsset(context, "prophets_full.json")
        if (json != null) { try { val type = object : TypeToken<List<ProphetModel>>() {}.type; return Gson().fromJson(json, type) } catch (_: Exception) {} }
        return (1..25).map { i -> ProphetModel(id = i, name = listOf("Adam","Idris","Nuh","Hud","Saleh","Ibrahim","Lut","Ismail","Ishaq","Yaqub","Yusuf","Ayyub","Shu'ayb","Musa","Harun","Dawud","Sulayman","Ilyas","Al-Yasa","Yunus","Dhul-Kifl","Zakariya","Yahya","Isa","Muhammad")[i-1]+" (A.S.)", arabicName = "نبي الله", pages = (1..5).map { p -> "${listOf("Adam","Idris","Nuh","Hud","Saleh","Ibrahim","Lut","Ismail","Ishaq","Yaqub","Yusuf","Ayyub","Shu'ayb","Musa","Harun","Dawud","Sulayman","Ilyas","Al-Yasa","Yunus","Dhul-Kifl","Zakariya","Yahya","Isa","Muhammad")[i-1]} Page $p - Wiladat, Dawat, Mojza, Sabr, Akhiri Paigham. Full story 200 words." }) }
    }
    fun loadQuran(context: Context): List<QuranVerseOrPage> {
        val json = readAsset(context, "quran_114.json")
        if (json != null) { try { val type = object : TypeToken<List<QuranVerseOrPage>>() {}.type; return Gson().fromJson(json, type) } catch (_: Exception) {} }
        return (1..604).map { i -> QuranVerseOrPage(i, "Surah #"+i, "بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ - $i", "Allah ke naam se Page $i", "Allah ke naam se Page $i Hindi", "In the name of Allah Page $i", "Allah chya navane Page $i", "Allah de naam naal Page $i") }
    }
    fun loadScientists(context: Context): List<ScientistModel> {
        val json = readAsset(context, "scientists_100.json")
        if (json != null) { try { val type = object : TypeToken<List<ScientistModel>>() {}.type; return Gson().fromJson(json, type) } catch (_: Exception) {} }
        return (1..100).map { i -> ScientistModel(i, "Scientist #$i", "Islamic Golden Era", "Optics", "Pioneered research") }
    }
}

class MainActivity : ComponentActivity(), TextToSpeech.OnInitListener {
    private var tts: TextToSpeech? = null
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        val db = AzadKahaniDatabase.getInstance(this)
        CoroutineScope(Dispatchers.IO).launch { if (db.sherDao().getCount() == 0) db.sherDao().insertAll(AssetDataRepository.loadShers(this@MainActivity)) }
        tts = TextToSpeech(this, this)
        setContent {
            MaterialTheme(colorScheme = darkColorScheme(primary = Color(0xFF00E676), background = Color(0xFF0A0F0D), surface = Color(0xFF131A17))) {
                Surface(modifier = Modifier.fillMaxSize(), color = Color(0xFF0A0F0D)) { AzadKahaniApp(db, tts) }
            }
        }
    }
    override fun onInit(status: Int) { if (status == TextToSpeech.SUCCESS) { tts?.language = Locale("hi","IN"); tts?.setSpeechRate(0.9f) } }
    override fun onDestroy() { tts?.stop(); tts?.shutdown(); super.onDestroy() }
}

enum class AzadTab(val title: String, val icon: ImageVector) { HOME("Home", Icons.Default.Home), REELS("Reels", Icons.Default.PlayArrow), KITAB("Kitab", Icons.Default.MenuBook), DUKAAN("Dukaan", Icons.Default.Storefront), SETTING("Jarvis", Icons.Default.SmartToy) }

@Composable
fun AzadKahaniApp(db: AzadKahaniDatabase, tts: TextToSpeech?) {
    var selectedTab by remember { mutableStateOf(AzadTab.HOME) }
    BackHandler(enabled = selectedTab != AzadTab.HOME) { selectedTab = AzadTab.HOME }
    Box(modifier = Modifier.fillMaxSize()) {
        Scaffold(bottomBar = {
            NavigationBar(containerColor = Color(0xFF131A17)) {
                AzadTab.values().forEach { tab ->
                    NavigationBarItem(icon = { Icon(tab.icon, contentDescription = tab.title) }, label = { Text(tab.title, fontSize = 11.sp) }, selected = selectedTab == tab, colors = NavigationBarItemDefaults.colors(selectedIconColor = Color(0xFF00E676), selectedTextColor = Color(0xFF00E676), unselectedIconColor = Color.Gray, unselectedTextColor = Color.Gray, indicatorColor = Color(0xFF1E2E25)), onClick = { selectedTab = tab })
                }
            }
        }) { paddingValues ->
            Box(modifier = Modifier.fillMaxSize().padding(paddingValues)) {
                when (selectedTab) {
                    AzadTab.HOME -> HomeShersScreen(db)
                    AzadTab.REELS -> ReelsScreen()
                    AzadTab.KITAB -> KitabScreen()
                    AzadTab.DUKAAN -> DukaanScreen(db)
                    AzadTab.SETTING -> JarvisScreen(tts)
                }
            }
        }
        FloatingJarvisBhaiya(tts)
    }
}

@OptIn(UnstableApi::class)
@Composable
fun ReelsScreen() {
    val context = LocalContext.current
    val reels = remember { mutableStateListOf(ReelData("1","https://commondatastorage.googleapis.com/gtv-videos-bucket/sample/ForBiggerBlazes.mp4","AZAD Reel #1",820), ReelData("2","https://commondatastorage.googleapis.com/gtv-videos-bucket/sample/ForBiggerEscapes.mp4","Khet Paani sukoon",1540)) }
    val launcher = rememberLauncherForActivityResult(ActivityResultContracts.GetContent()) { uri -> uri?.let { reels.add(0, ReelData(UUID.randomUUID().toString(), it.toString(), "My Uploaded Reel", 0)) } }
    Box(modifier = Modifier.fillMaxSize()) {
        val pagerState = rememberPagerState(pageCount = { reels.size })
        VerticalPager(state = pagerState, modifier = Modifier.fillMaxSize().background(Color.Black)) { index ->
            val reel = reels[index]; var isLiked by remember { mutableStateOf(false) }; var likes by remember { mutableIntStateOf(reel.likes) }
            val exoPlayer = remember { ExoPlayer.Builder(context).build().apply { setMediaItem(MediaItem.fromUri(reel.videoUrl)); repeatMode = ExoPlayer.REPEAT_MODE_ONE; prepare(); play() } }
            DisposableEffect(Unit) { onDispose { exoPlayer.release() } }
            Box(modifier = Modifier.fillMaxSize()) {
                AndroidView(factory = { ctx -> PlayerView(ctx).apply { player = exoPlayer; useController = false; layoutParams = FrameLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT) } }, modifier = Modifier.fillMaxSize())
                Column(modifier = Modifier.align(Alignment.BottomStart).padding(20.dp).fillMaxWidth(0.75f)) { Text(reel.title, color = Color.White, fontSize = 16.sp, fontWeight = FontWeight.SemiBold) }
                Column(modifier = Modifier.align(Alignment.BottomEnd).padding(end = 16.dp, bottom = 40.dp), horizontalAlignment = Alignment.CenterHorizontally) {
                    IconButton(onClick = { isLiked = !isLiked; likes = if (isLiked) likes+1 else likes-1 }) { Icon(Icons.Default.Favorite, contentDescription = "Like", tint = if (isLiked) Color.Red else Color.White, modifier = Modifier.size(34.dp)) }
                    Text("$likes", color = Color.White, fontSize = 12.sp)
                }
            }
        }
        FloatingActionButton(onClick = { launcher.launch("video/*") }, modifier = Modifier.align(Alignment.TopEnd).padding(16.dp), containerColor = Color(0xFF00E676)) { Icon(Icons.Default.Add, contentDescription = "Upload Reel") }
    }
}

        AlertDialog(onDismissRequest = { showAddDialog=false }, title = { Text("Dukaan par Saaman") }, text = { Column(verticalArrangement = Arrangement.spacedBy(8.dp)) { OutlinedTextField(value = name, onValueChange = { name=it }, label = { Text("Product Name") }); OutlinedTextField(value = rateStr, onValueChange = { rateStr=it }, label = { Text("Rate") }); OutlinedTextField(value = photoUri, onValueChange = { photoUri=it }, label = { Text("Photo Link") }) } }, confirmButton = { Button(onClick = { val rate = rateStr.toDoubleOrNull()?:0.0; if(name.isNotBlank()){ val id=UUID.randomUUID().toString(); val newProduct=ProductEntity(id,name,rate,photoUri); scope.launch { db.productDao().insert(newProduct) }; try{ FirebaseFirestore.getInstance().collection("dukaan_products").document(id).set(hashMapOf("id" to id,"name" to name,"rate" to rate,"photoUri" to photoUri)) }catch(_:Exception){}; showAddDialog=false } }) { Text("Upload") } }, dismissButton = { TextButton(onClick = { showAddDialog=false }) { Text("Cancel") } })

        lastCommand = command
        isListening = false
        scope.launch {
            if (command.contains("jarvis", true) || command.contains("bhaiya", true) || command.contains("party", true) || command.contains("task", true)) {

                    speak("Yes sir, boliye kya kaam hai. Aapke do pending task baki hain. Unhe main abhi complete karta hoon.")

                    Spacer(Modifier.height(8.dp))
                    Box(modifier = Modifier.size(80.dp).clip(CircleShape).background(Brush.radialGradient(listOf(Color(0xFF00E5FF), Color(0xFF0A0F0D)))).align(Alignment.CenterHorizontally), contentAlignment = Alignment.Center) { Text("●", color = Color.White, fontSize = 30.sp) }

                Column(modifier = Modifier.padding(10.dp)) {

            Column(modifier = Modifier.padding(16.dp)) {
                Text("Last Command: $lastCommand", color = Color(0xFF00E676), fontSize = 14.sp)
                Spacer(Modifier.height(12.dp))
                Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                    Button(onClick = {
                        isListening = true
                        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply { putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM); putExtra(RecognizerIntent.EXTRA_LANGUAGE, "hi-IN"); putExtra(RecognizerIntent.EXTRA_PROMPT, "Bolo sir...") }

        }
